216 auto const valid_index = [](Int_precision const index) {

227 [[nodiscard]] inline auto finite_incident_cells(
228 Delaunay const& triangulation, Edge_handle const& edge)
229 -> std::optional<Cell_container>
230 {
231 if (!is_well_formed_edge(edge) || triangulation.dimension() != 3 ||
232 !triangulation.tds().is_edge(edge.first, edge.second, edge.third))
233 {
234 return std::nullopt;
235 }
236
237 auto circulator = triangulation.incident_cells(edge, edge.first);
238 Cell_container incident_cells;
239 do
240 { // NOLINT(cppcoreguidelines-avoid-do-while)
241 if (triangulation.is_infinite(circulator)) { return std::nullopt; }
242 incident_cells.emplace_back(circulator);
243 }
244 while (++circulator != edge.first);
245 return incident_cells;
246 }

530 [[nodiscard]] inline auto detail::prepare_two_three(
531 Delaunay const& triangulation, Cell_handle const& candidate)
532 -> std::expected<ApplicableTwoThreeMove, MoveError>
533 {
534 using enum move_tracker::MoveType;
535 if (candidate == nullptr || triangulation.dimension() != 3 ||
536 !triangulation.tds().is_cell(candidate))
537 {
538 return move_error(MoveFailure::INVALID_TOPOLOGY, TWO_THREE);
539 }
540 if (!foliated_triangulations::is_cell_type_correct<3>(candidate) ||
541 foliated_triangulations::expected_cell_type<3>(candidate) !=
542 CellType::TWO_TWO)
543 {
544 return move_error(MoveFailure::CAUSAL_INVALIDITY, TWO_THREE);
545 }
546
547 std::array facet_indices{0, 1, 2, 3};
548 std::ranges::sort(facet_indices, [&](auto const left, auto const right) {
549 return detail::point_less(candidate->vertex(left)->point(),
550 candidate->vertex(right)->point());
551 });
552
553 for (auto const i : facet_indices)
554 {
555 auto const neighbor = candidate->neighbor(i);
556 if (triangulation.is_infinite(neighbor)) { continue; }
557
558 auto const neighbor_type =
559 foliated_triangulations::expected_cell_type<3>(neighbor);
560 if (!foliated_triangulations::is_cell_type_correct<3>(neighbor) ||
561 (neighbor_type != CellType::THREE_ONE &&
562 neighbor_type != CellType::ONE_THREE))
563 {
564 continue;
565 }
566
567 // A causal (2,3) move must replace the facet with a timelike edge.
568 // CGAL also permits topological flips that create a spacelike edge.
569 auto const mirror_index = neighbor->index(candidate);
570 auto const first_time =
571 static_cast<long long>(candidate->vertex(i)->info());
572 auto const second_time =
573 static_cast<long long>(neighbor->vertex(mirror_index)->info());
574 auto const time_difference = first_time > second_time
575 ? first_time - second_time
576 : second_time - first_time;
577 if (time_difference != 1) { continue; }
578
579 return ApplicableTwoThreeMove{canonical_cell_points(candidate),
580 candidate->vertex(i)->point()};
581 }
582
583 return move_error(MoveFailure::CAUSAL_INVALIDITY, TWO_THREE);
584 }

591 [[nodiscard]] inline auto detail::execute(Delaunay& triangulation,
592 ApplicableTwoThreeMove const& move)
593 -> Execution
594 {
595 using enum move_tracker::MoveType;
596 auto const cell = resolve_cell(triangulation, move.m_cell);
597 auto const opposite = resolve_vertex(triangulation, move.m_opposite);
598 if (!cell || !opposite)
599 {
600 return move_error(MoveFailure::STALE_CANDIDATE, TWO_THREE);
601 }
602
603 auto opposite_index = -1;
604 for (auto index = 0; index < 4; ++index)
605 {
606 if ((*cell)->vertex(index) == *opposite)
607 {
608 opposite_index = index;
609 break;
610 }
611 }
612 if (opposite_index < 0)
613 {
614 return move_error(MoveFailure::STALE_CANDIDATE, TWO_THREE);
615 }
616 if (!triangulation.flip(*cell, opposite_index))
617 {
618 return move_error(MoveFailure::EXECUTION_FAILURE, TWO_THREE);
619 }
620 return {};
621 }

649 template <std::uniform_random_bit_generator Generator>
650 [[nodiscard]] inline auto do_23_move(Manifold const& t_manifold,
651 Generator& generator) -> Expected
652 {
653 Delaunay triangulation{t_manifold.delaunay_snapshot()};
654 auto two_two = foliated_triangulations::filter_cells<3>(
655 foliated_triangulations::collect_cells<3>(triangulation),
656 CellType::TWO_TWO);
657 detail::canonicalize(two_two);
658 // Shuffle the container to create a random sequence of (2,2) cells
659 std::ranges::shuffle(two_two, generator);
660 if (two_two.empty())
661 {
662 return detail::move_error(MoveFailure::NO_CANDIDATE,
663 move_tracker::MoveType::TWO_THREE);
664 }
665
666 auto last_error =
667 MoveError{.category = MoveFailure::NO_CANDIDATE,
668 .requested_move = move_tracker::MoveType::TWO_THREE};
669 for (auto const& cell : two_two)
670 {
671 auto const prepared = detail::prepare_two_three(triangulation, cell);
672 if (!prepared)
673 {
674 last_error = prepared.error();
675 continue;
676 }
677 auto const executed = detail::execute(triangulation, *prepared);
678 if (executed)
679 {
680 return detail::make_manifold(std::move(triangulation), t_manifold);
681 }
682 last_error = executed.error();
683 }
684 return std::unexpected{last_error};
685 }

719 [[nodiscard]] inline auto is_32_movable(Delaunay const& triangulation,
720 Edge_handle const& candidate)
721 -> bool
722 {
723 auto const incident_cells =
724 finite_incident_cells(triangulation, candidate);
725 if (!incident_cells || incident_cells->size() != 3) { return false; }
726
727 auto const first_time = static_cast<long long>(
728 candidate.first->vertex(candidate.second)->info());
729 auto const second_time = static_cast<long long>(
730 candidate.first->vertex(candidate.third)->info());
731 auto const time_difference = first_time > second_time
732 ? first_time - second_time
733 : second_time - first_time;
734 if (time_difference != 1) { return false; }
735
736 if (!std::ranges::all_of(*incident_cells, [](auto const& cell) {
737 return foliated_triangulations::is_cell_type_correct<3>(cell);
738 }))
739 {
740 return false;
741 }
742
743 auto const cell_type_count = [&](CellType const type) {
744 return std::ranges::count_if(*incident_cells, [&](auto const& cell) {
745 return foliated_triangulations::expected_cell_type<3>(cell) == type;
746 });
747 };
748 auto const incident_31 = cell_type_count(CellType::THREE_ONE);
749 auto const incident_22 = cell_type_count(CellType::TWO_TWO);
750 auto const incident_13 = cell_type_count(CellType::ONE_THREE);
751 return incident_22 == 2 && ((incident_31 == 1 && incident_13 == 0) ||
752 (incident_31 == 0 && incident_13 == 1));
753 }

964 template <typename Post_mutation_validator>
965 requires std::predicate<Post_mutation_validator&, Delaunay const&>
966 [[nodiscard]] inline auto detail::execute(
967 Delaunay& triangulation, ApplicableTwoSixMove const& move,
968 Post_mutation_validator post_mutation_validator) -> Execution
969 {
970 using enum move_tracker::MoveType;
971 static constexpr auto incident_cell_count = std::size_t{6};
972
973 auto const bottom = resolve_cell(triangulation, move.bottom_points());
974 auto const opposite = resolve_vertex(triangulation, move.opposite_point());
975 if (!bottom || !opposite)
976 {
977 return move_error(MoveFailure::STALE_CANDIDATE, TWO_SIX);
978 }
979
980 auto common_face_index = -1;
981 for (auto index = 0; index < 4; ++index)
982 {
983 if ((*bottom)->vertex(index) == *opposite)
984 {
985 common_face_index = index;
986 break;
987 }
988 }
989 if (common_face_index < 0)
990 {
991 return move_error(MoveFailure::STALE_CANDIDATE, TWO_SIX);
992 }
993
994 auto const first = (common_face_index + 1) % 4;
995 auto const second = (common_face_index + 2) % 4;
996 auto const third = (common_face_index + 3) % 4;
997 auto const v_1 = (*bottom)->vertex(first);
998 auto const v_2 = (*bottom)->vertex(second);
999 auto const v_3 = (*bottom)->vertex(third);
1000 auto const center =
1001 triangulation.tds().insert_in_facet(*bottom, common_face_index);
1002
1003 Cell_container incident_cells;
1004 triangulation.tds().incident_cells(center,
1005 std::back_inserter(incident_cells));
1006 if (incident_cells.size() != incident_cell_count ||
1007 !std::ranges::all_of(incident_cells,
1008 [&triangulation](auto const& cell) {
1009 return triangulation.tds().is_cell(cell);
1010 }))
1011 {
1012 return move_error(MoveFailure::INVARIANT_VIOLATION, TWO_SIX);
1013 }
1014
1015 std::array face_points{v_1->point(), v_2->point(), v_3->point()};
1016 std::ranges::sort(face_points, point_less);
1017 center->set_point(
1018 CGAL::centroid(face_points[0], face_points[1], face_points[2]));
1019 center->info() = v_1->info();
1020
1021 if (!post_mutation_validator(static_cast<Delaunay const&>(triangulation)) ||
1022 !triangulation.tds().is_valid(center, true, 1))
1023 {
1024 return move_error(MoveFailure::INVARIANT_VIOLATION, TWO_SIX);
1025 }
1026 return {};
1027 }

1148 [[nodiscard]] inline auto detail::is_62_movable(
1149 Delaunay const& triangulation, Vertex_handle const& candidate) -> bool
1150 {
1151 if (triangulation.dimension() != 3) { return false; }
1152
1153 if (!triangulation.tds().is_vertex(candidate)) { return false; }
1154
1155 // We must have 5 incident edges to have 6 incident cells
1156 if (auto incident_edges = triangulation.degree(candidate);
1157 incident_edges != 5) // NOLINT
1158 {
1159 return false;
1160 }
1161
1162 // Obtain all incident cells
1163 Cell_container incident_cells;
1164 triangulation.tds().incident_cells(candidate,
1165 std::back_inserter(incident_cells));
1166
1167 // We must have 6 cells incident to the vertex to make a (6,2) move
1168 if (incident_cells.size() != 6) // NOLINT
1169 {
1170 return false;
1171 }
1172
1173 // Check that none of the incident cells are infinite
1174 for (auto const& cell : incident_cells)
1175 {
1176 if (triangulation.is_infinite(cell)) { return false; }
1177 }
1178
1179 auto const cell_type_count = [&](CellType const type) {
1180 return std::ranges::count_if(incident_cells, [&](auto const& cell) {
1181 return foliated_triangulations::expected_cell_type<3>(cell) == type;
1182 });
1183 };
1184 auto const incident_31 = cell_type_count(CellType::THREE_ONE);
1185 auto const incident_22 = cell_type_count(CellType::TWO_TWO);
1186 auto const incident_13 = cell_type_count(CellType::ONE_THREE);
1187
1188 // All cells should be causally classified and carry matching metadata.
1189 if (incident_13 + incident_22 + incident_31 != 6 || // NOLINT
1190 !std::ranges::all_of(incident_cells, [](auto const& cell) {
1191 return foliated_triangulations::is_cell_type_correct<3>(cell);
1192 }))
1193 {
1194 return false;
1195 }
1196
1197 return incident_31 == 3 && incident_22 == 0 && incident_13 == 3;
1198
1199 } // find_62_moves()

1503 auto const second = candidate.first->vertex(candidate.third);

1505 Vertex_handle bottom = nullptr;

1511 if (vertex == first || vertex == second) { continue; }

1513 if (bottom == nullptr || vertex_precedes(vertex, bottom))
1514 {
1515 bottom = vertex;
1516 }
1517 }
1518 }

1521 top == second || bottom == first || bottom == second)
1522 {
1523 return move_error(MoveFailure::CAUSAL_INVALIDITY, FOUR_FOUR);
1524 }

1527 return std::ranges::count_if(
1528 *incident_cells,
1529 [&](Cell_handle const cell) { return cell->has_vertex(vertex); });
1530 };

1532 {
1533 return move_error(MoveFailure::CAUSAL_INVALIDITY, FOUR_FOUR);
1534 }

1537 top->point(), bottom->point()};

1601 template <typename Post_mutation_validator>
1602 requires std::predicate<Post_mutation_validator&, Delaunay const&>
1603 [[nodiscard]] inline auto detail::execute(
1604 Delaunay const& source_triangulation, ApplicableFourFourMove const& move,
1605 Post_mutation_validator post_mutation_validator)
1606 -> std::expected<Delaunay, MoveError>
1607 {
1608 using enum move_tracker::MoveType;
1609 Delaunay triangulation{source_triangulation};
1610 auto const edge = resolve_edge(triangulation, move.edge_points());
1611 auto const top = resolve_vertex(triangulation, move.top_point());
1612 auto const bottom = resolve_vertex(triangulation, move.bottom_point());
1613 if (!edge || !top || !bottom)
1614 {
1615 return move_error(MoveFailure::STALE_CANDIDATE, FOUR_FOUR);
1616 }
1617
1618 auto const pivot_from_1 = edge->first->vertex(edge->second);
1619 auto const pivot_from_2 = edge->first->vertex(edge->third);
1620
1621 // A 3D 4-to-4 bistellar move is the composition of CGAL's checked TDS
1622 // 2-to-3 facet flip and checked 3-to-2 edge flip. The TDS operations are
1623 // intentional: either geometrically checked Triangulation_3 facet flip
1624 // can reject the transient, degenerate intermediate even when the final
1625 // 4-to-4 configuration is valid. Flipping a facet that contains the old
1626 // edge and either boundary vertex creates the new diagonal; the old edge
1627 // then has degree three and can be removed by the second flip.
1628 Cell_handle boundary_facet_cell = nullptr;
1629 int pivot_from_1_index{};
1630 int pivot_from_2_index{};
1631 int boundary_index{};
1632 if (!triangulation.is_facet(pivot_from_1, pivot_from_2, *bottom,
1633 boundary_facet_cell, pivot_from_1_index,
1634 pivot_from_2_index, boundary_index))
1635 {
1636 return move_error(MoveFailure::STALE_CANDIDATE, FOUR_FOUR);
1637 }
1638 constexpr auto cell_index_sum = 0 + 1 + 2 + 3;
1639 auto const boundary_facet_index = cell_index_sum - pivot_from_1_index -
1640 pivot_from_2_index - boundary_index;
1641 if (!triangulation.tds().flip(
1642 Delaunay::Facet{boundary_facet_cell, boundary_facet_index}))
1643 {
1644 return move_error(MoveFailure::EXECUTION_FAILURE, FOUR_FOUR);
1645 }
1646
1647 Cell_handle old_edge_cell = nullptr;
1648 int old_edge_first_index{};
1649 int old_edge_second_index{};
1650 if (!triangulation.is_edge(pivot_from_1, pivot_from_2, old_edge_cell,
1651 old_edge_first_index, old_edge_second_index) ||
1652 !triangulation.tds().flip(Delaunay::Edge{
1653 old_edge_cell, old_edge_first_index, old_edge_second_index}))
1654 {
1655 return move_error(MoveFailure::EXECUTION_FAILURE, FOUR_FOUR);
1656 }
1657
1658 if (!post_mutation_validator(static_cast<Delaunay const&>(triangulation)) ||
1659 !triangulation.tds().is_valid())
1660 {
1661 return move_error(MoveFailure::INVARIANT_VIOLATION, FOUR_FOUR);
1662 }
1663
1664 for (auto const cell : triangulation.finite_cell_handles())
1665 {
1666 cell->info() = static_cast<Int_precision>(
1667 foliated_triangulations::expected_cell_type<3>(cell));
1668 }
1669
1670 return triangulation;
1671 } // execute()
