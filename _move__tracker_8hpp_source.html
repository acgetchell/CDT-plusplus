60 [[nodiscard]] constexpr auto move_from_index(
61 std::size_t const move_choice) noexcept -> std::optional<MoveType>
62 {
63 using enum MoveType;
64 constexpr std::array moves{TWO_THREE, THREE_TWO, TWO_SIX, SIX_TWO,
65 FOUR_FOUR};
66 static_assert(moves.size() == NUMBER_OF_3D_MOVES);
67 if (move_choice >= moves.size()) { return std::nullopt; }
68 return moves[move_choice];
69 } // move_from_index

72 template <std::uniform_random_bit_generator Generator>
73 [[nodiscard]] inline auto generate_random_move_3(Generator& generator)
74 -> MoveType
75 {
76 std::uniform_int_distribution<int> distribution{
77 0, static_cast<int>(NUMBER_OF_3D_MOVES - 1)};
78 auto const move_choice = distribution(generator);
79 return *move_from_index(static_cast<std::size_t>(move_choice));
80 } // generate_random_move_3
