105 static void record_transition(
106 RunStatistics& statistics, move_tracker::MoveType const move,
107 ergodic_moves::MoveOutcome const outcome) noexcept
108 {
109 auto const append = [&statistics](std::uint8_t const value) {
110 statistics.transition_trace ^= value;
111 statistics.transition_trace *= 1099511628211ULL;
112 };
113 append(static_cast<std::uint8_t>(move));
114 append(static_cast<std::uint8_t>(outcome));
115 ++statistics.transition_count;
116 }

348 [[nodiscard]] static auto proposal_probability(
349 Geometry<ManifoldType::dimension> const& geometry,
350 move_tracker::MoveType const move) -> mpfr_values::Value
351 {
352 auto const sites = proposal_site_count(geometry, move);
353 if (sites <= 0) { return mpfr_values::zero(); }
354 auto const move_count =
355 mpfr_values::from_integer(move_tracker::NUMBER_OF_3D_MOVES);
356 auto const site_count = mpfr_values::from_integer(sites);
357 auto const denominator = mpfr_values::multiply(move_count, site_count);
358 return mpfr_values::divide(mpfr_values::from_integer(1), denominator);
359 }

371 [[nodiscard]] static auto hastings_ratio(
372 Geometry<ManifoldType::dimension> const& current,
373 Geometry<ManifoldType::dimension> const& proposed,
374 move_tracker::MoveType const move) -> mpfr_values::Value
375 {
376 auto const forward = proposal_probability(current, move);
377 auto const reverse_type = reverse_move(move);
378 if (!reverse_type)
379 {
380 throw std::invalid_argument{"Cannot reverse an unknown move type."};
381 }
382 auto const reverse = proposal_probability(proposed, *reverse_type);
383 if (mpfr_zero_p(forward.fr()) != 0 || mpfr_zero_p(reverse.fr()) != 0)
384 {
385 throw std::logic_error(
386 "A successful reversible proposal must have nonzero forward and reverse probabilities.");
387 }
388 return mpfr_values::divide(reverse, forward);
389 }

397 [[nodiscard]] auto action_ratio(
398 Geometry<ManifoldType::dimension> const& current,
399 Geometry<ManifoldType::dimension> const& proposed) const
400 -> mpfr_values::Value
401 {
402 auto const current_action = s3_action::s3_bulk_action(
403 current.N1_TL, current.N3_31_13, current.N3_22, m_parameters);
404 auto const proposed_action = s3_action::s3_bulk_action(
405 proposed.N1_TL, proposed.N3_31_13, proposed.N3_22, m_parameters);
406 return mpfr_values::exponential(
407 mpfr_values::subtract(current_action, proposed_action));
408 }

416 [[nodiscard]] auto acceptance_probability(
417 Geometry<ManifoldType::dimension> const& current,
418 Geometry<ManifoldType::dimension> const& proposed,
419 move_tracker::MoveType const move) const -> mpfr_values::Value
420 {
421 auto const ratio =
422 mpfr_values::multiply(hastings_ratio(current, proposed, move),
423 action_ratio(current, proposed));
424 auto const one = mpfr_values::from_integer(1);
425 return mpfr_cmp(ratio.fr(), one.fr()) < 0 ? ratio : one;
426 }

477 auto resolve_transition(ManifoldType& current,
478 CommandResults& command_results,
479 RunStatistics& statistics,
480 move_tracker::MoveType const move,
481 long double const trial_value)
482 -> ergodic_moves::MoveOutcome
483 {
484 if (!std::isfinite(trial_value) || trial_value < 0.0L ||
485 trial_value > 1.0L)
486 {
487 throw std::invalid_argument("MH trial value must lie in [0, 1].");
488 }
489
490 if (!reverse_move(move))
491 {
492 return ergodic_moves::outcome_from(ergodic_moves::MoveError{
493 .category = ergodic_moves::MoveFailure::UNKNOWN_MOVE,
494 .requested_move = move});
495 }
496
497 statistics.geometry = current.geometry();
498 ++statistics.proposed[move];
499 ++command_results.attempted[move];
500
501 auto candidate = propose_candidate(current, move);
502 if (!candidate)
503 {
504 ++command_results.failed[move];
505 ++statistics.rejected[move];
506 auto const outcome = ergodic_moves::outcome_from(candidate.error());
507 record_transition(statistics, move, outcome);
508 return outcome;
509 }
510 if (!ergodic_moves::detail::check_move(current, *candidate, move))
511 {
512 ++command_results.failed[move];
513 ++statistics.rejected[move];
514 record_transition(statistics, move,
515 ergodic_moves::MoveOutcome::EXECUTION_FAILED);
516 return ergodic_moves::MoveOutcome::EXECUTION_FAILED;
517 }
518
519 ++command_results.succeeded[move];
520 auto const probability = acceptance_probability(
521 statistics.geometry, candidate->geometry(), move);
522 if (mpfr_cmp_ld(probability.fr(), trial_value) >= 0)
523 {
524 swap(*candidate, current);
525 statistics.geometry = current.geometry();
526 ++statistics.accepted[move];
527 record_transition(statistics, move,
528 ergodic_moves::MoveOutcome::METROPOLIS_ACCEPTED);
529 return ergodic_moves::MoveOutcome::METROPOLIS_ACCEPTED;
530 }
531
532 ++statistics.rejected[move];
533 record_transition(statistics, move,
534 ergodic_moves::MoveOutcome::METROPOLIS_REJECTED);
535 return ergodic_moves::MoveOutcome::METROPOLIS_REJECTED;
536 }

549 [[nodiscard]] auto execute_pass(ManifoldType current,
550 RunStatistics statistics,
551 Int_precision const attempts) -> PassResult
552 {
553 auto command_results = CommandResults{};
554 for (auto move_attempt = Int_precision{0}; move_attempt < attempts;
555 ++move_attempt)
556 {
557 static_cast<void>(
558 sample_transition(current, command_results, statistics));
559 }
560 return {.manifold = std::move(current),
561 .command_results = std::move(command_results),
562 .strategy_state = std::move(statistics)};
563 }
