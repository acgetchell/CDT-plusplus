105 static void record_transition(
106 RunStatistics& statistics, move_tracker::MoveType const move,
107 ergodic_moves::MoveOutcome const outcome) noexcept
108 {
109 auto const append = [&statistics](std::uint8_t const value) {
110 statistics.transition_trace ^= value;
111 statistics.transition_trace *= 1099511628211ULL;
112 };
113 append(static_cast<std::uint8_t>(move));
114 append(static_cast<std::uint8_t>(outcome));
115 ++statistics.transition_count;
116 }

311 [[nodiscard]] static auto proposal_probability(
312 Geometry<ManifoldType::dimension> const& geometry,
313 move_tracker::MoveType const move) -> mpfr_values::Value
314 {
315 auto const sites = proposal_site_count(geometry, move);
316 if (sites <= 0) { return mpfr_values::zero(); }
317 auto const move_count =
318 mpfr_values::from_integer(move_tracker::NUMBER_OF_3D_MOVES);
319 auto const site_count = mpfr_values::from_integer(sites);
320 auto const denominator = mpfr_values::multiply(move_count, site_count);
321 return mpfr_values::divide(mpfr_values::from_integer(1), denominator);
322 }

327 [[nodiscard]] static auto hastings_ratio(
328 Geometry<ManifoldType::dimension> const& current,
329 Geometry<ManifoldType::dimension> const& proposed,
330 move_tracker::MoveType const move) -> mpfr_values::Value
331 {
332 auto const forward = proposal_probability(current, move);
333 auto const reverse_type = reverse_move(move);
334 if (!reverse_type)
335 {
336 throw std::invalid_argument{"Cannot reverse an unknown move type."};
337 }
338 auto const reverse = proposal_probability(proposed, *reverse_type);
339 if (mpfr_zero_p(forward.fr()) != 0 || mpfr_zero_p(reverse.fr()) != 0)
340 {
341 throw std::logic_error(
342 "A successful reversible proposal must have nonzero forward and reverse probabilities.");
343 }
344 return mpfr_values::divide(reverse, forward);
345 }

350 [[nodiscard]] auto action_ratio(
351 Geometry<ManifoldType::dimension> const& current,
352 Geometry<ManifoldType::dimension> const& proposed) const
353 -> mpfr_values::Value
354 {
355 auto const current_action = s3_action::s3_bulk_action(
356 current.N1_TL, current.N3_31_13, current.N3_22, m_parameters);
357 auto const proposed_action = s3_action::s3_bulk_action(
358 proposed.N1_TL, proposed.N3_31_13, proposed.N3_22, m_parameters);
359 return mpfr_values::exponential(
360 mpfr_values::subtract(current_action, proposed_action));
361 }

364 [[nodiscard]] auto acceptance_probability(
365 Geometry<ManifoldType::dimension> const& current,
366 Geometry<ManifoldType::dimension> const& proposed,
367 move_tracker::MoveType const move) const -> mpfr_values::Value
368 {
369 auto const ratio =
370 mpfr_values::multiply(hastings_ratio(current, proposed, move),
371 action_ratio(current, proposed));
372 auto const one = mpfr_values::from_integer(1);
373 return mpfr_cmp(ratio.fr(), one.fr()) < 0 ? ratio : one;
374 }

425 auto attempt_transition(ManifoldType& current,
426 CommandResults& command_results,
427 RunStatistics& statistics,
428 move_tracker::MoveType const move,
429 long double const trial_value) -> bool
430 {
431 if (!std::isfinite(trial_value) || trial_value < 0.0L ||
432 trial_value > 1.0L)
433 {
434 throw std::invalid_argument("MH trial value must lie in [0, 1].");
435 }
436
437 statistics.geometry = current.geometry();
438 ++statistics.proposed[move];
439 ++command_results.attempted[move];
440
441 auto candidate = propose_candidate(current, move);
442 if (!candidate)
443 {
444 ++command_results.failed[move];
445 ++statistics.rejected[move];
446 record_transition(statistics, move,
447 ergodic_moves::outcome_from(candidate.error()));
448 return false;
449 }
450 if (!ergodic_moves::detail::check_move(current, *candidate, move))
451 {
452 ++command_results.failed[move];
453 ++statistics.rejected[move];
454 record_transition(statistics, move,
455 ergodic_moves::MoveOutcome::EXECUTION_FAILED);
456 return false;
457 }
458
459 ++command_results.succeeded[move];
460 auto const probability = acceptance_probability(
461 statistics.geometry, candidate->geometry(), move);
462 if (mpfr_cmp_ld(probability.fr(), trial_value) >= 0)
463 {
464 swap(*candidate, current);
465 statistics.geometry = current.geometry();
466 ++statistics.accepted[move];
467 record_transition(statistics, move,
468 ergodic_moves::MoveOutcome::METROPOLIS_ACCEPTED);
469 return true;
470 }
471
472 ++statistics.rejected[move];
473 record_transition(statistics, move,
474 ergodic_moves::MoveOutcome::METROPOLIS_REJECTED);
475 return false;
476 }

478 [[nodiscard]] auto execute_pass(ManifoldType current,
479 RunStatistics statistics,
480 Int_precision const attempts) -> PassResult
481 {
482 auto command_results = CommandResults{};
483 std::uniform_real_distribution<long double> acceptance_draw{0.0L, 1.0L};
484 for (auto move_attempt = Int_precision{0}; move_attempt < attempts;
485 ++move_attempt)
486 {
487 auto const move = move_tracker::generate_random_move_3(m_generator);
488 static_cast<void>(attempt_transition(current, command_results,
489 statistics, move,
490 acceptance_draw(m_generator)));
491 }
492 return {.manifold = std::move(current),
493 .command_results = std::move(command_results),
494 .strategy_state = std::move(statistics)};
495 }
